// insights.js — نصايح الأرباح والسيولة لمنصة تدفق // الاستخدام: ثم: TadaffuqInsights.load(supabase, document.getElementById('tips')) // عدّل أسماء الجداول والأعمدة من الكائن SCHEMA تحت حسب اللي عندك في Supabase. const TadaffuqInsights = (() => { const SCHEMA = { receivables: { table: 'receivables', amount: 'amount', issued: 'issue_date', due: 'due_date', paid: 'paid_date', party: 'customer' }, payables: { table: 'payables', amount: 'amount', due: 'due_date', paid: 'paid_date', party: 'supplier' }, transactions: { table: 'transactions', amount: 'amount', date: 'date', type: 'type', category: 'category' }, // type: 'income' | 'expense' inventory: { table: 'inventory', qty: 'qty', cost: 'unit_cost', lastMove: 'last_movement_date', name: 'item' }, cash: { table: 'cash_accounts', balance: 'balance' }, }; const DAY = 864e5; const today = () => new Date(); const d = (v) => (v ? new Date(v) : null); const sum = (a, f) => a.reduce((s, x) => s + (Number(f(x)) || 0), 0); const fmt = (n) => Math.round(n).toLocaleString('ar-EG'); function compute({ recv, pay, tx, inv, cash }) { const S = SCHEMA, now = today(); const inRange = (row, from, to) => { const t = d(row[S.transactions.date]); return t >= from && t < to; }; const t30 = new Date(now - 30 * DAY), t60 = new Date(now - 60 * DAY), t90 = new Date(now - 90 * DAY); const isType = (r, ty) => r[S.transactions.type] === ty; const amt = (r) => Number(r[S.transactions.amount]) || 0; const cur = tx.filter((r) => inRange(r, t30, now)); const prev = tx.filter((r) => inRange(r, t60, t30)); const inc = sum(cur.filter((r) => isType(r, 'income')), amt); const exp = sum(cur.filter((r) => isType(r, 'expense')), amt); const incP = sum(prev.filter((r) => isType(r, 'income')), amt); const expP = sum(prev.filter((r) => isType(r, 'expense')), amt); const cashNow = sum(cash, (r) => r[S.cash.balance]); // المديونيات المتأخرة عند العملاء const unpaidR = recv.filter((r) => !r[S.receivables.paid]); const overdue = unpaidR.filter((r) => d(r[S.receivables.due]) < now) .sort((a, b) => b[S.receivables.amount] - a[S.receivables.amount]); const overdueTotal = sum(overdue, (r) => r[S.receivables.amount]); // متوسط أيام التحصيل آخر 90 يوم const paidR = recv.filter((r) => r[S.receivables.paid] && d(r[S.receivables.paid]) >= t90); const dso = paidR.length ? paidR.reduce((s, r) => s + (d(r[S.receivables.paid]) - d(r[S.receivables.issued])) / DAY, 0) / paidR.length : null; // التزامات + متوقع تحصيله خلال 30 يوم const in30 = new Date(+now + 30 * DAY); const due30 = sum(pay.filter((r) => !r[S.payables.paid] && d(r[S.payables.due]) <= in30), (r) => r[S.payables.amount]); const collect30 = sum(unpaidR.filter((r) => d(r[S.receivables.due]) <= in30), (r) => r[S.receivables.amount]); // أعلى بنود المصروفات const byCat = {}; cur.filter((r) => isType(r, 'expense')).forEach((r) => { const c = r[S.transactions.category] || 'أخرى'; byCat[c] = (byCat[c] || 0) + amt(r); }); const topCats = Object.entries(byCat).sort((a, b) => b[1] - a[1]).slice(0, 3); // مخزون راكد 90 يوم const stale = inv.filter((r) => d(r[S.inventory.lastMove]) < t90); const staleValue = sum(stale, (r) => r[S.inventory.qty] * r[S.inventory.cost]); return { cashNow, inc, exp, incP, expP, margin: inc ? (inc - exp) / inc : 0, marginPrev: incP ? (incP - expP) / incP : 0, coverage: exp ? cashNow / exp : null, overdue, overdueTotal, dso, due30, collect30, topCats, stale, staleValue, }; } function buildTips(m) { const tips = []; const add = (priority, level, title, why, impact, action) => tips.push({ priority, level, title, why, impact, action }); if (m.cashNow + m.collect30 < m.due30) add(100, 'danger', 'فجوة سيولة قادمة', `التزامات 30 يوم (${fmt(m.due30)}) أكبر من النقدية + المتوقع تحصيله (${fmt(m.cashNow + m.collect30)})`, `العجز المتوقع ${fmt(m.due30 - m.cashNow - m.collect30)}`, 'اطلب تحصيل مبكر من أكبر العملاء، وفاوض الموردين على تأجيل جزء من السداد.'); if (m.coverage !== null && m.coverage < 2) add(90, 'danger', 'السيولة تغطي أقل من شهرين', `النقدية تكفي ${m.coverage.toFixed(1)} شهر من المصروفات`, 'خطر توقف التشغيل عند أي تأخير في التحصيل', 'أجّل المشتريات غير الضرورية وابنِ احتياطي لا يقل عن شهرين.'); if (m.overdueTotal > 0) add(80, 'warn', 'مديونيات متأخرة عند العملاء', `${m.overdue.length} فاتورة متأخرة بإجمالي ${fmt(m.overdueTotal)}`, `تحصيلها يرفع السيولة بـ ${fmt(m.overdueTotal)}`, `ابدأ بأكبر 3 مديونيات: ${m.overdue.slice(0, 3).map((r) => r[SCHEMA.receivables.party]).join('، ')}.`); if (m.dso !== null && m.dso > 45) add(70, 'warn', 'التحصيل بطيء', `متوسط التحصيل ${Math.round(m.dso)} يوم`, 'كل يوم تأخير يعني نقدية محبوسة', 'جرّب خصم سداد مبكر (مثلاً 2% للدفع خلال 10 أيام) أو اشترط دفعة مقدمة.'); if (m.marginPrev && m.margin < m.marginPrev - 0.02) add(75, 'warn', 'هامش الربح نزل', `من ${(m.marginPrev * 100).toFixed(1)}% إلى ${(m.margin * 100).toFixed(1)}%`, `فرق تقريبي ${fmt((m.marginPrev - m.margin) * m.inc)} في الشهر`, 'راجع هل السبب زيادة تكلفة ولا نزول أسعار البيع، وعدّل التسعير أو غيّر المورد.'); if (m.topCats.length) { const [c, v] = m.topCats[0]; add(60, 'info', 'أكبر بند مصروفات', `"${c}" = ${fmt(v)} (${m.exp ? Math.round((v / m.exp) * 100) : 0}% من المصروفات)`, `توفير 5% فيه = ${fmt(v * 0.05)} شهرياً`, 'قارن أسعار موردين بديلة أو تفاوض على عقد سنوي.'); } if (m.staleValue > 0) add(65, 'warn', 'مخزون راكد', `${m.stale.length} صنف لم يتحرك 90 يوم، قيمته ${fmt(m.staleValue)}`, 'رأس مال واقف ممكن يتحول لسيولة', 'اعمل عرض تخفيض أو باقات، وأوقف إعادة شرائها.'); if (m.coverage !== null && m.coverage > 6) add(40, 'info', 'سيولة زيادة عن الحاجة', `النقدية تغطي ${m.coverage.toFixed(1)} شهر`, 'فرصة ضائعة لو فضلت راكدة', 'فكّر في سداد التزام مكلف أو وديعة أو توسع مدروس.'); return tips.sort((a, b) => b.priority - a.priority); } // لو عاوز جيميني يصيغ النصايح بأسلوب بسيط: ابعت الناتج ده في الـ prompt const toPrompt = (tips) => 'أنت مستشار مالي. اشرح النصايح دي لصاحب مشروع بالعامية المصرية في 3 جمل لكل نصيحة، ولا تضف أرقام من عندك:\n' + JSON.stringify(tips.slice(0, 3), null, 1); function render(tips, el) { const color = { danger: '#c62828', warn: '#ef6c00', info: '#1565c0' }; el.dir = 'rtl'; el.innerHTML = tips.length ? tips.slice(0, 3).map((t) => `
${t.title}
📊 ${t.why}
💰 ${t.impact}
✅ ${t.action}
`).join('') + 'توصيات استرشادية مبنية على بياناتك، وليست قراراً نهائياً.' : '

الوضع المالي مستقر، لا توجد تنبيهات حالياً 👌

'; } async function load(supabase, el) { const get = async (t) => { const { data, error } = await supabase.from(t).select('*'); if (error) throw error; return data || []; }; const S = SCHEMA; const [recv, pay, tx, inv, cash] = await Promise.all( [S.receivables.table, S.payables.table, S.transactions.table, S.inventory.table, S.cash.table].map(get)); const tips = buildTips(compute({ recv, pay, tx, inv, cash })); render(tips, el); return tips; } return { load, compute, buildTips, render, toPrompt, SCHEMA }; })();
